import { agent, triggers } from "@helmr/sdk"
import { createAgentSession, SessionManager } from "@earendil-works/pi-coding-agent"
import { workspace } from "./computer"

export const coder = agent({
  id: "coder",
  template: workspace,

  async init({ session }) {
    const { session: pi } = await createAgentSession({
      cwd: "/app",
      sessionManager: SessionManager.continueRecent(
        "/app",
        `/var/lib/helmr/pi/${session.id}`,
      ),
    })
    return pi
  },

  async turn(turn, { initResult: pi }) {
    turn.onCancel(() => pi.abort())
    const request = turn.input.map(part => part.text).join("")
    await pi.prompt(`${request}\n\nPropose a plan. Do not edit files yet.`)

    // Ask a teammate before Pi changes anything.
    const { answer } = await turn.ask({
      type: "choice",
      prompt: pi.getLastAssistantText() ?? "Pi has a plan.",
      options: [
        { id: "apply", label: "Apply", value: "Apply the plan and run the tests." },
        { id: "skip", label: "Not now", value: "Leave the files unchanged." },
      ],
    })
    await pi.prompt(answer.selected[0]!.value)
    await turn.respond(pi.getLastAssistantText() ?? "")
    return { piSession: pi.sessionManager.getSessionId() }
  },

  triggers: [
    triggers.cron("weekday-deps", "0 9 * * 1-5", {
      timezone: "Asia/Tokyo",
      input: [{ type: "text", text: "Update dependencies." }],
    }),
  ],
})
