import { agent, triggers } from "@helmr/sdk"
import { createAgentSession, SessionManager } from "@earendil-works/pi-coding-agent"
import { workspace } from "./computer"

export const coder = agent({
  id: "coder",
  template: workspace,

  async init({ session }) {
    const { session: pi } = await createAgentSession({
      cwd: "/app",
      sessionManager: SessionManager.continueRecent("/app", `/var/lib/helmr/pi/${session.id}`),
    })
    return pi
  },

  async turn(turn, { initResult: pi }) {
    // Ask a teammate before choosing the scope of this task.
    const { answer } = await turn.ask({
      type: "choice",
      prompt: "Apply dependency updates, or inspect them first?",
      options: [
        { id: "apply", label: "Apply updates", value: "Apply the updates." },
        { id: "inspect", label: "Inspect first", value: "Inspect only; do not edit." },
      ],
    })
    const prompt = turn.input.map(part => part.text).join("")
    turn.onCancel(() => pi.abort())
    await pi.prompt(`${prompt}\n\n${answer.selected[0]!.value}`)
    await turn.respond(pi.getLastAssistantText() ?? "")
    return { piSession: pi.sessionManager.getSessionId() }
  },

  triggers: [
    triggers.cron("weekday-deps", "0 9 * * 1-5", {
      timezone: "Asia/Tokyo",
      input: [{ type: "text", text: "Update dependencies." }],
    }),
  ],
})
