import { agent, triggers } from "@helmr/sdk"
import { createOpencodeClient, createOpencodeServer } from "@opencode-ai/sdk/v2"
import { workspace } from "./computer"

export const coder = agent({
  id: "coder",
  template: workspace,

  async init() {
    const { url } = await createOpencodeServer()
    const opencode = createOpencodeClient({ baseUrl: url, directory: "/app" })
    const { data } = await opencode.session.create({}, { throwOnError: true })
    return { opencode, sessionID: data.id }
  },

  async turn(turn, { initResult: { opencode, sessionID } }) {
    // Ask a teammate before choosing the scope of this task.
    const { answer } = await turn.ask({
      type: "choice",
      prompt: "Apply dependency updates, or inspect them first?",
      options: [
        { id: "apply", label: "Apply updates", value: "Apply the updates." },
        { id: "inspect", label: "Inspect first", value: "Inspect only; do not edit." },
      ],
    })
    const prompt = turn.input.map(part => part.text).join("")
    const { data } = await opencode.session.prompt({
      sessionID,
      parts: [{ type: "text", text: `${prompt}\n\n${answer.selected[0]!.value}` }],
    }, { throwOnError: true })
    const reply = data.parts.map(part => part.type === "text" ? part.text : "")
    await turn.respond(reply.join(""))
    return { sessionID }
  },

  triggers: [
    triggers.cron("weekday-deps", "0 9 * * 1-5", {
      timezone: "Asia/Tokyo",
      input: [{ type: "text", text: "Update dependencies." }],
    }),
  ],
})
