import { template, image } from "@helmr/sdk"

export const workspace = template({
  id: "app",
  image: image("app")
    .from("node:24-bookworm")
    .run(["sh", "-c", "apt-get update && apt-get install -y ripgrep"]),
  resources: { cpu: 2, memory: "4GiB" },

  prepareSecrets: [{
    secret: "REGISTRY_TOKEN",
    env: { name: "REGISTRY_TOKEN", mode: "raw" },
  }],
  secrets: [{
    secret: "ANTHROPIC_API_KEY",
    env: {
      name: "ANTHROPIC_API_KEY",
      mode: "protected",
      allowedOrigins: ["https://api.anthropic.com"],
    },
  }],

  async prepare(build) {
    await build.exec(["git", "clone", "https://github.com/acme/app", "/app"])
    await build.exec(["npm", "ci"], { cwd: "/app" })
  },
})
